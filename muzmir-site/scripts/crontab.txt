# ЕДИНЫЙ CRONTAB САЙТА КЦ «МУЗЫКАЛЬНЫЙ МИР» — ИСТОЧНИК ПРАВДЫ.
#
# Этот файл — полный список заданий. Раньше их было два и оба неполные: здесь
# лежали 8 строк, в deploy/DEPLOY.md — 9 «полного набора», а реально в cron/
# двадцать с лишним рабочих скриптов. Ключевые (launch_scheduler — исполнитель
# волн запуска, monthly_launch — открытие нового месяца) не значились нигде,
# и при восстановлении crontab после переустановки сервера кампания просто
# не запускалась бы, причём молча.
#
# Установка целиком (ЗАМЕЩАЕТ текущий crontab):
#   crontab /var/www/muzmir/scripts/crontab.txt
# Проверка расхождений без замены:
#   php /var/www/muzmir/scripts/check_crontab.php
#
# ЧАСОВОЙ ПОЯС. Все времена ниже — МОСКОВСКИЕ. CRON_TZ ниже фиксирует это на
# уровне самого крона, чтобы часы не «поехали» при переустановке сервера с
# другой зоной по умолчанию. PHP переводится в Москву отдельно (config.php).
CRON_TZ=Europe/Moscow
MAILTO=""

# ── Каждую минуту ───────────────────────────────────────────────────────────
# Очередь писем: личные — сразу, массовые — по дневной норме и в окне 09–18.
* * * * * php /var/www/muzmir/cron/process_newsletter_queue.php >/dev/null 2>&1
# Исполнитель волн запуска (пульт «Запуск»). БЕЗ НЕГО КАМПАНИЯ НЕ ИДЁТ ВООБЩЕ.
* * * * * php /var/www/muzmir/cron/launch_scheduler.php >/dev/null 2>&1
# Дипломы: генерация и отправка по расписанию рабочих дней.
* * * * * php /var/www/muzmir/cron/send_diplomas.php >>/var/log/muzmir_send_diplomas.log 2>&1
# Дожим неоплаченных заявок (первое письмо через 3 минуты после подачи).
* * * * * php /var/www/muzmir/cron/dunning.php >/dev/null 2>&1
# Личные сообщения сообщества ВК (очередь vk_dm_queue) — только открытые диалоги.
* * * * * php /var/www/muzmir/cron/vk_dm_worker.php >/dev/null 2>&1

# ── Чаще раза в час ─────────────────────────────────────────────────────────
# Сверка платежей с ЮKassa (pull-реконсиляция — работает и без вебхуков).
*/2 * * * * php /var/www/muzmir/cron/reconcile_payments.php >/dev/null 2>&1
# Доступность сайта.
*/5 * * * * php /var/www/muzmir/cron/health_check.php >/dev/null 2>&1
# Отложенные афиши в соцсети.
*/5 * * * * php /var/www/muzmir/cron/publish_scheduled_posts.php >/dev/null 2>&1
# Ответы на обращения, пришедшие в нерабочее время («отвечу, как начнётся день»).
*/10 * * * * php /var/www/muzmir/cron/chat_offhours_flush.php >/dev/null 2>&1
# Закрытие «повисших» диалогов бота ВК.
*/10 * * * * php /var/www/muzmir/cron/vk_bot_close.php >/dev/null 2>&1
# Разбор возвратов почты (bounce).
*/30 * * * * php /var/www/muzmir/cron/process_bounces.php >> /var/www/muzmir/data/logs/bounces.log 2>&1

# ── Раз в час ───────────────────────────────────────────────────────────────
# Маркетинговые волны конкурсов (открытие приёма, «3 дня», «последний день»).
0 * * * * php /var/www/muzmir/cron/mailings.php >/dev/null 2>&1

0 * * * * php /var/www/muzmir/cron/club_billing.php >/dev/null 2>&1

0 3 * * * php /var/www/muzmir/cron/daily_backup.php >/dev/null 2>&1

0 9 * * * php /var/www/muzmir/cron/check_competitions_dates.php >/dev/null 2>&1

# Полный бэкап (БД + uploads + дипломы), ночь с воскресенья на понедельник.
0 4 * * 1 php /var/www/muzmir/cron/weekly_backup.php >/dev/null 2>&1

0 9 1 * * php /var/www/muzmir/cron/monthly_launch.php >> /var/www/muzmir/data/logs/launch.log 2>&1

0 8 1 * * php /var/www/muzmir/cron/monthly_report.php >/dev/null 2>&1

0 10 28 * * php /var/www/muzmir/cron/publish_results_vk.php >/dev/null 2>&1

0 6 5 1 * php /var/www/muzmir/cron/annual_report.php >/dev/null 2>&1
